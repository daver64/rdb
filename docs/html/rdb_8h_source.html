342 void executeQuery(SQLResults* results, const std::string& sql) {
343 try {
344 results->clear();
345 auto stmt = db_->prepare(sql);
346
347 int column_count = sqlite3_column_count(stmt->stmt_);
348
349 while (stmt->step()) {
350 SQLRow row;
351 for (int i = 0; i < column_count; i++) {
352 const char* col_name = sqlite3_column_name(stmt->stmt_, i);
353 const char* col_text = reinterpret_cast<const char*>(
354 sqlite3_column_text(stmt->stmt_, i)
355 );
356 row[col_name ? col_name : ""] = col_text ? col_text : "";
357 }
358 results->results.push_back(row);
359 }
360
361 results->num_rows = results->results.size();
362 results->num_fields = column_count;
363 results->num_tuples = results->num_rows;
364 results->row_iterator = results->results.begin();
365
366 } catch (const SQLiteException& e) {
367 results->error_message = e.what();
368 results->num_rows = 0;
369 results->num_fields = 0;
370 }
