201 static std::string translatePlaceholders(const std::string& sql) {
202 std::string translated;
203 int parameter = 0;
204 bool quoted = false;
205 for (size_t i = 0; i < sql.size(); ++i) {
206 if (sql[i] == '\'' && (i == 0 || sql[i - 1] != '\\')) quoted = !quoted;
207 if (sql[i] == '?' && !quoted) translated += "$" + std::to_string(++parameter);
208 else translated += sql[i];
209 }
210 return translated;
211 }

216 if (!result_ || (PQresultStatus(result_) != PGRES_TUPLES_OK && PQresultStatus(result_) != PGRES_COMMAND_OK)) {

301 std::memset(&parameters_[index - 1], 0, sizeof(MYSQL_BIND));
302 parameters_[index - 1].buffer_type = MYSQL_TYPE_STRING;
303 parameters_[index - 1].buffer = const_cast<char*>(values_[index - 1].data());
304 parameters_[index - 1].buffer_length = static_cast<unsigned long>(values_[index - 1].size());

313 const unsigned int count = mysql_num_fields(metadata_);

316 if (mysql_stmt_bind_result(statement_, &results_[0]) != 0 || mysql_stmt_store_result(statement_) != 0) throw UnifiedException(mysql_stmt_error(statement_));
317 executed_ = true;
